\section{Modelo de minimização de custos}

O modelo busca \textbf{minimizar o custo logístico total} de transporte entre plantas da rede Ball Corporation, respeitando capacidades produtivas e logísticas e garantindo o atendimento à demanda de cada SKU por destino e período.

\[
  \boxed{
    \min \; Z = \sum_{i \in I} \sum_{l \in L} \sum_{j \in P} \sum_{t \in T}
    ct_{\,\phi(l),\,j} \cdot \frac{x_{i,\,l,\,j,\,t}}{200{.}000}
  }
\]

\begin{center}
  \small\color{darkgray}
  Sujeito às restrições de capacidade logística, atendimento à demanda,\\
  capacidade produtiva e não-negatividade detalhadas nas seções seguintes.
\end{center}

% ============================================================
\section{Custos (Parâmetros Logísticos)}

Seja o par $(o, j)$ representando origem $o$ e destino $j$, com $o \in P$ e $j \in P$:

\begin{align*}
    cb_{o,j} &: \text{custo de combustível da rota } o \to j \\
    cp_{o,j} &: \text{custo de pedágio da rota } o \to j \\
    cl_{o,j} &: \text{custo de mão de obra da rota } o \to j \\
    ct_{o,j} &= cb_{o,j} + cp_{o,j} + cl_{o,j} \qquad \text{(custo total do frete por caminhão)}
\end{align*}

\textbf{Fluxo interno:} quando $o = j$, fixamos $ct_{o,j} = 0$.

\section{Conjuntos (Índices)}

\begin{center}

  \midrule

  $L$ & Conjunto de linhas de produção ativas, indexado por $l$ & 30 linhas \\
  $P$ & Conjunto de plantas (fábricas), indexado por $p$ & 10 plantas \\
  $T$ & Conjunto de períodos $(t_{01}, t_{02}, \ldots, t_{15})$, indexado por $t$ & 15 períodos \\
  \bottomrule
\end{tabular}
\end{center}

% ============================================================
\section{Funções Auxiliares}

\begin{align*}
    \gamma(i, l, t) &= v \quad \text{(volume de latas demandado)} \\
    \phi(l)         &= p \quad \text{(mapeamento hierárquico: linha } l \mapsto \text{planta } p\text{)} \\
    \psi(o,j)       &= ct_{o,j} \quad \text{com } ct_{o,j} = 0 \text{ se } o = j
\end{align*}

% ============================================================
\section{Variável de Decisão}

\[
    x_{i,\,l,\,j,\,t} \;\in\; \mathbb{R}_+
\]

\begin{center}
  \small
  \textit{Volume (latas) do produto $i$, produzido na linha $l$, destinado à planta $j$, no período $t$.}
\end{center}

% ============================================================
\section{Função Objetivo}

Minimizar o custo logístico total da malha:

\[
    \min \; Z = \sum_{i \in I} \sum_{l \in L} \sum_{j \in P} \sum_{t \in T}
    ct_{\,\phi(l),\,j} \cdot \frac{x_{i,\,l,\,j,\,t}}{200{.}000}
\]

onde $\dfrac{x_{i,l,j,t}}{200{.}000}$ representa o número de caminhões necessários
(capacidade de $25\text{ pallets} \times 8{.}000\text{ latas} = 200{.}000\text{ latas/caminhão}$).

% ============================================================
\section{Restrições}

Variáveis auxiliares utilizadas nas restrições:

\begin{itemize}[leftmargin=1.5em]
  \item $n_{i,\,j,\,t}$: demanda do produto $i$ com destino $j$ no período $t$.
  \item $r_{i,\,l,\,t}$: taxa de produção do produto $i$ na linha $l$ no período $t$.
\end{itemize}

\vspace{8pt}

\restricao{R1 — Capacidade logística de escoamento.}{%
  80 caminhões/dia por planta, cada um com capacidade de 200.000 latas:
  \[
    \sum_{i \in I}\; \sum_{\substack{l \in L \\ \phi(l) = p}}\; \sum_{j \in P}
    \frac{x_{i,\,l,\,j,\,t}}{200{.}000}
    \;\leq\; 80
    \qquad \forall\, p \in P,\; \forall\, t \in T
  \]}

\restricao{R2 — Atendimento à demanda.}{%
  A produção deve suprir a demanda $n_{i,j,t}$ de cada SKU por planta e período:
  \[
    \sum_{l \in L} x_{i,\,l,\,j,\,t}
    \;\geq\; n_{i,\,j,\,t}
    \qquad \forall\, i \in I,\; \forall\, j \in P,\; \forall\, t \in T
  \]}

\restricao{R3 — Capacidade produtiva.}{%
  As horas de uso da linha não ultrapassam o turno de 24 horas:
  \[
    \sum_{i \in I}\; \sum_{j \in P}
    \frac{x_{i,\,l,\,j,\,t}}{r_{i,\,l,\,t}}
    \;\leq\; 24
    \qquad \forall\, l \in L,\; \forall\, t \in T
  \]}

\restricao{R4 — Não-negatividade.}{%
  \[
    x_{i,\,l,\,j,\,t} \;\geq\; 0
    \qquad \forall\, i \in I,\; l \in L,\; j \in P,\; t \in T
  \]}

\vspace{12pt}
{\color{accentgray}\rule{\linewidth}{0.5pt}}
\begin{center}
  \small\color{darkgray}
  Dimensão do problema: $|I| \times |L| \times |P| \times |T| = 30 \times 30 \times 10 \times 15 = \mathbf{135{.}000}$ variáveis de decisão.
\end{center}
